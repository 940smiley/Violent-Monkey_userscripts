    'https://www.claimdepot.com/',
    'https://www.openclassactions.com/',
    'https://www.classaction.com/'
];

// Load or initialize URLs to scan
let scanUrls = GM_getValue('scanUrls', defaultUrls);

// Register menu command to add URLs
GM_registerMenuCommand('Add URL to Scan List', () => {
    const newUrl = prompt('Enter a new URL to scan for settlements:');
    if (newUrl) {
        if (!scanUrls.includes(newUrl)) {
            scanUrls.push(newUrl);
            GM_setValue('scanUrls', scanUrls);
            alert('URL added to scan list. Reload the page to apply.');
        } else {
            alert('URL already in scan list.');
        }
    }
});
